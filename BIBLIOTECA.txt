    ID_LIVRO INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NOME VARCHAR(254) NOT NULL,
    AUTOR VARCHAR(254),
    LIVRO VARCHAR(254),
    DATAPUBLICACAO INTEGER
)
